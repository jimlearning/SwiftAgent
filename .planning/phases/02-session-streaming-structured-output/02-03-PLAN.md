    WAIT — this reveals a design issue. The RESEARCH.md code assumes the channel yields to the continuation AND the agent loop can inspect what happened. But RuntimeGenerationChannel yields directly to the consumer — the loop can't "collect" events from it.

    CORRECTED DESIGN: Two different channel types are needed:
    - RuntimeGenerationChannel (created in Plan 02-01): yields directly to continuation, used when the consumer IS the loop consumer (streamResponse path)
    - A collecting channel (internal to AgentRuntimeImpl): records events locally, used when the loop needs to inspect events before deciding to continue

    For the streaming path, we need a "tee" pattern: the executor emits events, the channel yields them to the consumer AND records tool calls for the loop to process.

    SIMPLEST CORRECT APPROACH: Don't tee. Instead:

    1. Create a collecting channel (private actor inside AgentRuntimeImpl.swift)
    2. Pass it to the executor
    3. After the executor finishes, iterate collected events:
       - For text/thinking events: yield them to continuation
       - For tool calls: execute them, append results to transcript, set turnComplete = false (re-loop)
       - For turnCompleted: set turnComplete = true, yield the event to continuation

    ```
    /// Does NOT yield to a continuation directly — the loop yields after inspection.
    private actor CollectingChannel: GenerationChannel {
        private(set) var events: [SessionEvent] = []
        private var isFinished: Bool = false
        private var accumulatedText: String = ""
        private var accumulatedThinking: String = ""

        func send(textDelta: String) async {
            guard !isFinished else { return }
            accumulatedText = textDelta
            events.append(.textDelta(accumulatedText))
        }

        func send(thinkingDelta: String) async {
            guard !isFinished else { return }
            accumulatedThinking = thinkingDelta
            events.append(.thinkingDelta(accumulatedThinking))
        }

        func send(toolCallRequest id: String, name: String, input: Data) async {
            guard !isFinished else { return }
            events.append(.toolCallRequested(id: id, name: name, input: input))
        }

        func send(toolCallCompleted id: String, output: ToolOutputValue) async {
            guard !isFinished else { return }
            events.append(.toolCallCompleted(id: id, output: output, isError: false))
        }

        func complete(stopReason: String?, usage: Usage?) async {
            guard !isFinished else { return }
            isFinished = true
            events.append(.turnCompleted(usage: usage, stopReason: stopReason))
        }

        func fail(with error: AgentRuntimeError) async {
            guard !isFinished else { return }
            isFinished = true
            events.append(.error(error))
        }
    }
    ```

    ```
    private func runLoop(
        prompt: String,
        continuation: AsyncThrowingStream<SessionEvent, Error>.Continuation
    ) async throws {
        transcript.entries.append(.prompt(prompt))

        let toolDefs = await toolEngine.getAllDefinitions()

        var turnComplete = false
        var iterationCount = 0
        let maxIterations = 50
        var responseText = ""

        while !turnComplete && iterationCount < maxIterations {
            iterationCount += 1

            let channel = CollectingChannel()
            let executor = modelProvider.makeExecutor()

            try await executor.respond(
                to: transcript,
                tools: toolDefs,
                options: GenerationOptions(),
                streamingInto: channel
            )

            let events = await channel.events

            for event in events {
                switch event {
                case .textDelta(let text):
                    responseText = text  // snapshot replacement
                    continuation.yield(.textDelta(text))

                case .thinkingDelta(let thinking):
                    continuation.yield(.thinkingDelta(thinking))

                case .toolCallRequested(let id, let name, let input):
                    // Yield to consumer, then execute
                    continuation.yield(.toolCallRequested(id: id, name: name, input: input))
                    let output = try await executeTool(name: name, input: input)
                    transcript.entries.append(.toolOutput(id: id, output: output.stringValue, isError: false))
                    continuation.yield(.toolCallCompleted(id: id, output: output, isError: false))
                    // Note: tool execution means we loop back to re-prompt the model
                    turnComplete = false

                case .toolCallCompleted:
                    break  // handled above with toolCallRequested

                case .turnCompleted(let usage, let stopReason):
                    continuation.yield(.turnCompleted(usage: usage, stopReason: stopReason))
                    turnComplete = true

                case .error(let err):
                    continuation.yield(.error(err))
                    throw err
                }
            }

            if turnComplete {
                break
            }
        }

        // After turn completes: append response to transcript, update memory
        if !responseText.isEmpty {
            transcript.entries.append(.response(responseText))
        }
        try? await memoryStore.store(key: "latest", namespace: "sessions", value: transcript)
    }
    ```
